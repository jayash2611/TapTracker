          package com.jayash.taptracker;

          import android.app.PendingIntent;
          import android.appwidget.AppWidgetManager;
          import android.appwidget.AppWidgetProvider;
          import android.content.ComponentName;
          import android.content.Context;
          import android.content.Intent;
          import android.view.View;
          import android.widget.RemoteViews;
          import org.json.JSONArray;
          import org.json.JSONObject;
          import java.util.Calendar;

          public class TapWidgetProvider extends AppWidgetProvider {
              static final String ADD = "com.jayash.taptracker.ADD";
              static final String UNDO = "com.jayash.taptracker.UNDO";
              static final int[] BTN = {R.id.b0, R.id.b1, R.id.b2};

              static JSONObject load(Context c) {
                  try {
                      String s = c.getSharedPreferences("tap", Context.MODE_PRIVATE).getString("state", "");
                      if (s.length() > 0) return new JSONObject(s);
                  } catch (Exception e) { }
                  return null;
              }

              static void store(Context c, JSONObject s) {
                  c.getSharedPreferences("tap", Context.MODE_PRIVATE).edit().putString("state", s.toString()).apply();
              }

              @Override
              public void onReceive(Context c, Intent i) {
                  super.onReceive(c, i);
                  String a = i.getAction();
                  try {
                      if (ADD.equals(a)) add(c, i.getIntExtra("idx", 0));
                      else if (UNDO.equals(a)) undo(c);
                      else return;
                  } catch (Exception e) { }
                  refresh(c);
              }

              static void add(Context c, int idx) throws Exception {
                  JSONObject s = load(c);
                  if (s == null) return;
                  JSONArray items = s.getJSONArray("items");
                  if (idx >= items.length()) return;
                  JSONObject item = items.getJSONObject(idx);
                  JSONArray logs = s.optJSONArray("logs");
                  if (logs == null) { logs = new JSONArray(); s.put("logs", logs); }
                  int next = s.optInt("next", 1);
                  JSONObject l = new JSONObject();
                  l.put("id", next);
                  l.put("item", item.getInt("id"));
                  l.put("name", item.optString("name"));
                  if (item.has("k")) l.put("k", item.getString("k"));
                  l.put("ts", System.currentTimeMillis());
                  l.put("tag", "");
                  l.put("price", item.optDouble("price", 0));
                  logs.put(l);
                  s.put("next", next + 1);
                  store(c, s);
              }

              static void undo(Context c) throws Exception {
                  JSONObject s = load(c);
                  if (s == null) return;
                  JSONArray logs = s.optJSONArray("logs");
                  if (logs == null || logs.length() == 0) return;
                  logs.remove(logs.length() - 1);
                  store(c, s);
              }

              static void refresh(Context c) {
                  AppWidgetManager m = AppWidgetManager.getInstance(c);
                  int[] ids = m.getAppWidgetIds(new ComponentName(c, TapWidgetProvider.class));
                  for (int id : ids) m.updateAppWidget(id, views(c));
              }

              @Override
              public void onUpdate(Context c, AppWidgetManager m, int[] ids) {
                  for (int id : ids) m.updateAppWidget(id, views(c));
              }

              static PendingIntent pi(Context c, int code, String action, int idx) {
                  Intent i = new Intent(c, TapWidgetProvider.class);
                  i.setAction(action);
                  i.putExtra("idx", idx);
                  return PendingIntent.getBroadcast(c, code, i, PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE);
              }

              static RemoteViews views(Context c) {
                  RemoteViews v = new RemoteViews(c.getPackageName(), R.layout.tap_widget);
                  Intent open = c.getPackageManager().getLaunchIntentForPackage(c.getPackageName());
                  if (open != null) {
                      v.setOnClickPendingIntent(R.id.total, PendingIntent.getActivity(c, 99, open, PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE));
                  }
                  v.setOnClickPendingIntent(R.id.ub, pi(c, 50, UNDO, 0));
                  JSONObject s = load(c);
                  if (s == null) {
                      v.setTextViewText(R.id.total, "Open Tap Tracker");
                      v.setTextViewText(R.id.count, "");
                      for (int b : BTN) v.setViewVisibility(b, View.GONE);
                      return v;
                  }
                  double money = 0;
                  int count = 0;
                  try {
                      Calendar cal = Calendar.getInstance();
                      cal.set(Calendar.HOUR_OF_DAY, 0);
                      cal.set(Calendar.MINUTE, 0);
                      cal.set(Calendar.SECOND, 0);
                      cal.set(Calendar.MILLISECOND, 0);
                      long t0 = cal.getTimeInMillis();
                      JSONArray logs = s.optJSONArray("logs");
                      if (logs != null) {
                          for (int k = 0; k < logs.length(); k++) {
                              JSONObject l = logs.getJSONObject(k);
                              if (l.getLong("ts") >= t0) { count++; money += l.optDouble("price", 0); }
                          }
                      }
                      JSONArray items = s.getJSONArray("items");
                      for (int k = 0; k < 3; k++) {
                          if (k < items.length()) {
                              v.setTextViewText(BTN[k], items.getJSONObject(k).optString("icon", "+"));
                              v.setViewVisibility(BTN[k], View.VISIBLE);
                              v.setOnClickPendingIntent(BTN[k], pi(c, 10 + k, ADD, k));
                          } else {
                              v.setViewVisibility(BTN[k], View.GONE);
                          }
                      }
                  } catch (Exception e) { }
                  v.setTextViewText(R.id.total, "\u20B9" + Math.round(money));
                  v.setTextViewText(R.id.count, count + " today");
                  return v;
              }
          }
          EOF
